cmake_minimum_required(VERSION 3.20)
project(bot_face_tests C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)
set(UI ${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/components/bot_ui)
set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/components/bot_core)
set(MOTION ${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/components/bot_motion)
add_library(face_motion STATIC ${UI}/face_motion.c ${UI}/face_geometry.c ${UI}/face_reaction.c ${MOTION}/bot_motion.c)
target_include_directories(face_motion PUBLIC ${UI}/include ${CORE}/include ${MOTION}/include)
target_link_libraries(face_motion PUBLIC m)
add_executable(test_face_motion test_face_motion.c)
target_link_libraries(test_face_motion PRIVATE face_motion)
add_executable(test_ui_lifecycle test_ui_lifecycle.c stubs/fake_lvgl.c
    ${UI}/ui.c ${UI}/face.c ${CORE}/gesture.c ${CORE}/bot_router.c)
target_include_directories(test_ui_lifecycle PRIVATE stubs)
target_link_libraries(test_ui_lifecycle PRIVATE face_motion)
add_executable(dump_face_frames ../../tools/dump_face_frames.c)
target_link_libraries(dump_face_frames PRIVATE face_motion)
enable_testing()
add_test(NAME face_motion COMMAND test_face_motion)
add_test(NAME ui_lifecycle COMMAND test_ui_lifecycle)
